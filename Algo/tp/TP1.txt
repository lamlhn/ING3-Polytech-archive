int positif_3(int nb1, int nb2)
{
    if(nb1 > nb2){
    	return nb2;
    }
    return nb1;
}

int main()
{
    int a, b, min;
    
    printf("Saisissez deux entiers: \n");
    scanf("%d%d", &a, &b);
    
    min = positif_3(a, b);
    printf("Minimum: %d\n", min);
    
    return 0;
}

#include <stdio.h>

float moyenne(int n)
{
	int val;
	float moy;
 	printf("Saisissez des series entiers: \n");
    scanf("%d", &val);
    
    moy = val;
 	for(int i = 2; i <=n; i++)
	{
		scanf("%d", &val);
		moy += val;
	}
	moy = moy*1.0/n;
    return moy;
}

int main()
{
    int n, moy;
    
    printf("Saisissez n: \n");
    scanf("%d", &n);
    
    moy = moyenne(n);
    printf("Moyenne de %d entier: %d\n", n, moy);
    
    return 0;
}

void maxPos()
{
	int pos, n;
	float max, val;
 	printf("Saisissez un entier:");
    scanf("%f", &val);
    n = 1;
    max = val;
	pos = n;

    scanf("%f", &val);
    while(val != 0)
    {
    	n += 1;

    		max = val;
    		pos = n;

    }
